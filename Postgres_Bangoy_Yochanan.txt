DROP SCHEMA public CASCADE;
CREATE SCHEMA public;

-- 1. create tables

CREATE TABLE dim_product (
	product_id 	 INT PRIMARY KEY,	 
	product_name VARCHAR(100) NOT NULL,
	product_line VARCHAR(100),
	price        NUMERIC NOT NULL
);

CREATE TABLE dim_customer (
    customer_id     INT PRIMARY KEY,
    customer_name   VARCHAR(100) NOT NULL,
    customer_type   VARCHAR(50) NOT NULL,
    gender          VARCHAR(50) NOT NULL
);

CREATE TABLE dim_branch (
    branch_id       INT PRIMARY KEY,
    branch_name     VARCHAR(50) NOT NULL,
    branch_city     VARCHAR(50) NOT NULL
);

CREATE TABLE dim_payment (
    payment_id      INTEGER PRIMARY KEY,
    payment_type    VARCHAR(50) NOT NULL
);

CREATE TABLE fact_sales (
	invoice_id   VARCHAR(100) PRIMARY KEY,
 	branch_id    INT NOT NULL REFERENCES dim_branch(branch_id),
 	customer_id	 INT NOT NULL REFERENCES dim_customer(customer_id), 
 	product_id   INT NOT NULL REFERENCES dim_product(product_id),
 	quantity     INT NOT NULL,
 	date		 DATE,
 	time		 TIME,
 	payment		 FLOAT,
 	rating		 FLOAT	 
);

-- 1. Total revenue by branch city
SELECT b.branch_city,
    SUM(f.quantity * p.price) AS total_revenue
FROM fact_sales f
JOIN dim_branch b
    ON f.branch_id = b.branch_id
JOIN dim_product p
    ON f.product_id = p.product_id
GROUP BY b.branch_city
ORDER BY total_revenue DESC;

-- 2. Top 5 customers who spends the most
SELECT c.customer_id, c.customer_name,
    SUM(f.quantity * p.price) AS total_spending
FROM fact_sales f
JOIN dim_customer c
    ON f.customer_id = c.customer_id
JOIN dim_product p
    ON f.product_id = p.product_id
GROUP BY c.customer_id, c.customer_name
ORDER BY total_spending DESC
LIMIT 5;

-- 3. Total quantity sold by product line
SELECT p.product_line,
	SUM(fs.quantity) AS total_quantity_sold
FROM fact_sales fs
JOIN dim_product p
	ON fs.product_id = p.product_id
GROUP BY p.product_line
ORDER BY total_quantity_sold DESC;

-- 4. Show the distribution of payment type used
SELECT dp.payment_type,
	COUNT(*) AS number_transaction
FROM fact_sales fs
JOIN dim_payment dp
	ON fs.payment = dp.payment_id
GROUP BY dp.payment_type
ORDER BY number_transaction DESC;

-- 5. Average rating per product line
SELECT p.product_line,
	ROUND(AVG(fs.rating):: numeric, 2) AS average_rating
FROM fact_sales fs
JOIN dim_product p
	ON fs.product_id = p.product_id 
GROUP BY p.product_line
ORDER BY average_rating DESC;
